    function syncLogoToSocialThumbnail() {
      try {
        const svgEl = document.getElementById('hongly-svg-logo');
        if (!svgEl) return;
        const serializer = new XMLSerializer();
        const svgStr = serializer.serializeToString(svgEl);
        const encoded = 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(svgStr);
        const ogImg = document.getElementById('og-logo-thumb');
        const twImg = document.getElementById('twitter-logo-thumb');
        if (ogImg) ogImg.setAttribute('content', encoded);
        if (twImg) twImg.setAttribute('content', encoded);
      } catch (err) {
        /* Giữ nguyên fallback nếu trình duyệt giới hạn */
      }
    }
